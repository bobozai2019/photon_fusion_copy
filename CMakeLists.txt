cmake_minimum_required(VERSION 3.21)
project(netcode LANGUAGES C CXX)

include(CTest)
option(NETCODE_COCOS "Build the Cocos2d-x 4 adapter and car yard" OFF)
option(NETCODE_GODOT "Build the Godot 4 adapter and car yard" OFF)
set(COCOS2DX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../cocos2d-x-4" CACHE PATH "Cocos2d-x 4 source")

add_subdirectory(core)
if(BUILD_TESTING)
    add_subdirectory(tests/core)
endif()

if(NETCODE_COCOS OR NETCODE_GODOT)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The car yard requires Windows x64 and MSVC.")
    endif()
    add_subdirectory(examples/car-yard)
    if(NETCODE_COCOS)
        add_subdirectory(engineplatform/cocos2dx)
    endif()
    if(NETCODE_GODOT)
        add_subdirectory(engineplatform/godot)
    endif()
endif()
