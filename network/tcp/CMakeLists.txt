cmake_minimum_required(VERSION 3.21)
project(NetcodeTransportTcp VERSION 1.0.0 LANGUAGES CXX)

set(NETCODE_TRANSPORT_INCLUDE "" CACHE PATH "Installed Netcode transport SDK include directory")
if(NOT EXISTS "${NETCODE_TRANSPORT_INCLUDE}/netcode/transport_provider.h")
    message(FATAL_ERROR "Install network/api and set NETCODE_TRANSPORT_INCLUDE to its include directory.")
endif()

file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/provider.cpp" source_hash)
file(SHA256 "${NETCODE_TRANSPORT_INCLUDE}/netcode/transport_provider.h" api_hash)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS provider.cpp
    "${NETCODE_TRANSPORT_INCLUDE}/netcode/transport_provider.h")
string(SHA256 build_hash "${source_hash}|${api_hash}|${CMAKE_SYSTEM_NAME}|${CMAKE_CXX_COMPILER_VERSION}")
string(SUBSTRING "${build_hash}" 0 16 build_id)
add_library(netcode_transport_tcp MODULE provider.cpp)
target_compile_features(netcode_transport_tcp PRIVATE cxx_std_17)
target_include_directories(netcode_transport_tcp PRIVATE "${NETCODE_TRANSPORT_INCLUDE}")
target_compile_definitions(netcode_transport_tcp PRIVATE NC_TRANSPORT_BUILD_DLL NETCODE_TCP_BUILD_ID="${build_id}")
set_target_properties(netcode_transport_tcp PROPERTIES CXX_VISIBILITY_PRESET hidden PREFIX "")
if(WIN32)
    target_link_libraries(netcode_transport_tcp PRIVATE ws2_32)
endif()
install(TARGETS netcode_transport_tcp RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
install(FILES "${NETCODE_TRANSPORT_INCLUDE}/netcode/transport_provider.h" DESTINATION include/netcode)
