cmake_minimum_required(VERSION 3.21)
project(NetcodeTransportDirect VERSION 1.0.0 LANGUAGES CXX)

set(NETCODE_TRANSPORT_INCLUDE "" CACHE PATH "Installed Netcode transport SDK include directory")
if(NOT EXISTS "${NETCODE_TRANSPORT_INCLUDE}/netcode/transport_provider.h")
    message(FATAL_ERROR "Install network/api and set NETCODE_TRANSPORT_INCLUDE to its include directory.")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The Direct transport supports Windows and Linux.")
endif()
set(core "${CMAKE_CURRENT_SOURCE_DIR}/../../core")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/direct.cpp" direct_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/abi.cpp" abi_hash)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS direct.cpp abi.cpp)
string(SHA256 build_hash "${direct_hash}|${abi_hash}|${CMAKE_SYSTEM_NAME}|${CMAKE_CXX_COMPILER_VERSION}")
string(SUBSTRING "${build_hash}" 0 16 build_id)
add_library(netcode_transport_direct MODULE abi.cpp direct.cpp "${core}/transport/wire.cpp")
target_compile_features(netcode_transport_direct PRIVATE cxx_std_17)
target_include_directories(netcode_transport_direct PRIVATE "${NETCODE_TRANSPORT_INCLUDE}" "${core}")
target_compile_definitions(netcode_transport_direct PRIVATE NC_TRANSPORT_BUILD_DLL NETCODE_DIRECT_BUILD_ID="${build_id}")
set_target_properties(netcode_transport_direct PROPERTIES CXX_VISIBILITY_PRESET hidden PREFIX "")
if(WIN32)
    target_link_libraries(netcode_transport_direct PRIVATE ws2_32 bcrypt)
endif()
install(TARGETS netcode_transport_direct RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
