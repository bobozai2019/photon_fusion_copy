cmake_minimum_required(VERSION 3.21)
project(NetcodeCocos VERSION 0.1.0 LANGUAGES CXX)
include(CMakePackageConfigHelpers)
find_package(NetcodeCore 0.1.0 EXACT CONFIG REQUIRED)
set(COCOS2DX_ROOT "" CACHE PATH "Cocos2d-x 4 source or header tree")
if(NOT EXISTS "${COCOS2DX_ROOT}/cocos/2d/CCComponent.h")
    message(FATAL_ERROR "Set COCOS2DX_ROOT to the host's Cocos2d-x 4 tree")
endif()
option(NETCODE_COCOS_PHYSICS2D "Match the host CC_USE_PHYSICS" ON)
option(NETCODE_COCOS_PHYSICS3D "Match the host CC_USE_3D_PHYSICS" ON)
foreach(dimension IN ITEMS 2 3)
    if(NETCODE_COCOS_PHYSICS${dimension}D)
        set(physics${dimension}d 1)
    else()
        set(physics${dimension}d 0)
    endif()
endforeach()
configure_file(netcode_cocos_config.hpp.in netcode_cocos_config.hpp @ONLY)
add_library(netcode_cocos STATIC game_component.cpp)
set_target_properties(netcode_cocos PROPERTIES EXPORT_NAME Cocos)
target_include_directories(netcode_cocos PRIVATE "${COCOS2DX_ROOT}/cocos" "${COCOS2DX_ROOT}/cocos/platform"
    "$<$<PLATFORM_ID:Windows>:${COCOS2DX_ROOT}/cocos/platform/win32>")
target_include_directories(netcode_cocos INTERFACE "$<INSTALL_INTERFACE:include/netcode/cocos>")
netcode_build_id(platform_id "${CMAKE_CURRENT_SOURCE_DIR}")
string(SHA256 physics_id "${platform_id}/${physics2d}/${physics3d}")
string(SUBSTRING "${physics_id}" 0 16 platform_id)
target_compile_definitions(netcode_cocos PRIVATE CC_STATIC CC_USE_PHYSICS=${physics2d} CC_USE_3D_PHYSICS=${physics3d}
    NETCODE_PLATFORM_BUILD_ID="${platform_id}")
target_link_libraries(netcode_cocos PUBLIC Netcode::Direct Netcode::TransportHost "$<INSTALL_INTERFACE:cocos2d>")
set(NETCODE_TCP_MODULE "" CACHE FILEPATH "Built default Windows TCP provider")
if(NETCODE_TCP_MODULE)
    install(FILES "${NETCODE_TCP_MODULE}" DESTINATION bin RENAME netcode_transport_tcp.dll)
endif()
install(TARGETS netcode_cocos EXPORT NetcodeCocosTargets ARCHIVE DESTINATION lib)
install(FILES game_component.hpp replica_component.hpp forecast_body.hpp forecast_body_3d.hpp physics_island_3d.hpp
    "${CMAKE_CURRENT_BINARY_DIR}/netcode_cocos_config.hpp" DESTINATION include/netcode/cocos)
set(pkg lib/cmake/NetcodeCocos)
install(EXPORT NetcodeCocosTargets NAMESPACE Netcode:: DESTINATION ${pkg})
configure_package_config_file(NetcodeCocosConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/NetcodeCocosConfig.cmake" INSTALL_DESTINATION ${pkg})
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/NetcodeCocosConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/NetcodeCocosConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/NetcodeCocosConfigVersion.cmake" DESTINATION ${pkg})
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build-id.txt" "${NetcodeCore_BUILD_ID}/${platform_id}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build-id.txt" DESTINATION .)
