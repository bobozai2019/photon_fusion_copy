cmake_minimum_required(VERSION 3.21)
project(NetcodeUnity VERSION 0.1.0 LANGUAGES CXX)
find_package(NetcodeCore 0.1.0 EXACT CONFIG REQUIRED)
add_library(netcode_unity SHARED game.cpp)
target_link_libraries(netcode_unity PRIVATE Netcode::Direct Netcode::TransportHost)
netcode_build_id(platform_id "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(netcode_unity PRIVATE NETCODE_PLATFORM_BUILD_ID="${platform_id}")
install(TARGETS netcode_unity RUNTIME DESTINATION Runtime/Plugins/x86_64)
set(NETCODE_TCP_MODULE "" CACHE FILEPATH "Built default Windows TCP provider")
if(NETCODE_TCP_MODULE)
    install(FILES "${NETCODE_TCP_MODULE}" DESTINATION Runtime/Plugins/x86_64 RENAME netcode_transport_tcp.dll)
    install(FILES netcode_transport_tcp.dll.meta DESTINATION Runtime/Plugins/x86_64)
endif()
install(DIRECTORY managed/ DESTINATION Runtime)
install(DIRECTORY PluginMetadata/ DESTINATION Runtime/Plugins/x86_64)
install(DIRECTORY PackageMetadata/ DESTINATION .)
install(FILES package.json DESTINATION .)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build-id.txt" "${NetcodeCore_BUILD_ID}/${platform_id}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/build-id.txt" DESTINATION .)
